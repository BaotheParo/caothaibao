@Service
@RequiredArgsConstructor
public class RequestCoalescingService {
// Map tracking in-flight futures per idempotent key
private final ConcurrentHashMap<String, CompletableFuture<ProcessedResult>> inFlight = new ConcurrentHashMap<>();
public CompletableFuture<ProcessedResult> executeCoalesced(String key, Payload payload) {
return inFlight.computeIfAbsent(key, k -> {
return CompletableFuture.supplyAsync(() -> workerRouter.evaluate(payload))
.whenComplete((res, ex) -> inFlight.remove(k));
});
}
}
